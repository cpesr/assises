% ============================================================
% Titres de chapitre
% ============================================================

% Parties
\titleformat{\part}[display]
  {\titlefont\Huge\bfseries}
  {\centering\partname~\thepart}
  {12pt}
  {}

% Chapitres
\titleformat{\chapter}[display]
  {\titlefont}
  {\large\chaptername~\thechapter}
  {8pt}
  {\Huge\bfseries}

% Sections
\titleformat{\section}
  {\titlefont\Large\bfseries}
  {\thesection.}
  {0.5em}
  {}

% espace avant / après le titre
\titlespacing*{\chapter}
  {0pt}
  {0pt}
  {6pt}
